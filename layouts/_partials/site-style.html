{{- /* Переопределение partial темы ananke. Отличия от оригинала (themes/ananke/layouts/_partials/site-style.html): - Tachyons (_tachyons.css) больше НЕ подключается: вся вёрстка на Tailwind. - Из CSS темы остаются только «семантические» файлы: подсветка кода с кнопкой копирования, иконки соцсетей и стили встроенных шаблонов Hugo. - Tailwind собирается из assets/css/main.css через css.TailwindCSS внутри templates.Defer — так Hugo сначала отрисует все страницы, запишет hugo_stats.json, и только потом Tailwind сгенерирует нужные классы. */ -}} {{- /* 1. Оставшийся CSS темы ananke */ -}} {{- $theme_css := slice -}} {{- range slice "_code.css" "_hugo-internal-templates.css" "_social-icons.css" -}} {{- with resources.Get (printf "ananke/css/%s" .) -}} {{- $theme_css = $theme_css | append . -}} {{- end -}} {{- end -}} {{- with $theme_css -}} {{- $style := . | resources.Concat "css/theme.css" | css.Build (dict "minify" hugo.IsProduction) -}} {{- if hugo.IsProduction -}}{{- $style = $style | resources.Fingerprint -}}{{- end -}} {{- end }} {{- /* 2.
